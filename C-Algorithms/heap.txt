heap is a structure that is a complete binarytree we have min heap and max heap
for example in max heap the rule is that the parent is always bigger than its children
and all its descendants and the max number is the root of the tree. it is used in priority
queue where for example if we have multiple things to do we wanna do the thing with the highes priority first which is in the root . it is also used for sorting algorithm heapsort .
 void heapify(int * array,int n,int i){
int l=(2*i)+1;
int r=(2*i)+2;
int max=i;
if(l<n && array[l]>array[max]) max=l;
if(r<n && array[r]>array[max]) ,max=r;
if(i!=max){
int temp=array[i];
array[i]=array[max];
array[max]=temp;
heapify(array,n,max);
}


}

void heap_build(int* array,int n){
for(int i=(n/2)-1;i>=0;i--){

 heapify(array,n,i);}



}
void heapsort(int* array,int n){
heap_build(array,n);

for(int i=n-1;i>0;i--){
int temp=array[0];
array[0]=array[i];
array[i]=temp;
heapify(array,i,0);
}

}
